"""Hand-calculated contracts for the separate verification path."""
import unittest

import verify_chronological as verify


class IndependentVerificationTests(unittest.TestCase):
    def test_distinct_scores_below_float64_resolution_share_one_score(self):
        fitted = verify.pava([0.1, 0.1+5e-16, 0.8], [0, 1, 1])
        self.assertEqual(verify.interpolate(0.1, *fitted), 0.5)

    def test_weighted_duplicate_probability_pava(self):
        fitted = verify.pava([0.1, 0.2, 0.2, 0.3, 0.9], [0, 1, 0, 0, 1])
        self.assertAlmostEqual(verify.interpolate(0.2, *fitted), 1/3)
        self.assertAlmostEqual(verify.interpolate(0.3, *fitted), 1/3)
        self.assertAlmostEqual(verify.interpolate(0.15, *fitted), 1/6)

    def test_decreasing_outcomes_are_pooled(self):
        fitted = verify.pava([0.2, 0.8], [1, 0])
        self.assertEqual([verify.interpolate(p, *fitted) for p in (0., 0.5, 1.)], [0.5]*3)

    def test_single_unique_probability_and_endpoint_clipping(self):
        fitted = verify.pava([0.5, 0.5], [0, 1])
        self.assertEqual([verify.interpolate(p, *fitted) for p in (0., 0.5, 1.)], [0.5]*3)
        fitted = verify.pava([0.2, 0.8], [0, 1])
        self.assertAlmostEqual(verify.interpolate(0.5, *fitted), 0.5)
        self.assertEqual([verify.interpolate(p, *fitted) for p in (0., 1.)], [0., 1.])


if __name__ == "__main__":
    unittest.main()
