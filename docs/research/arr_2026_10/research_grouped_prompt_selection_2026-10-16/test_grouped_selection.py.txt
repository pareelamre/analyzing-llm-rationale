"""Synthetic contracts for exact-title-grouped selection; no study-data scores."""
import unittest

import grouped_selection as analysis


class GroupedSelectionTests(unittest.TestCase):
    def test_groups_are_disjoint_balanced_deterministic_and_order_invariant(self):
        titles = {q: f"title {q // 3}" for q in range(30)}
        folds = analysis.assign_group_folds(titles, 5, 42)
        self.assertEqual(folds, analysis.assign_group_folds(dict(reversed(list(titles.items()))), 5, 42))
        self.assertEqual(set(folds.values()), set(range(5)))
        self.assertEqual([sum(f == k for f in folds.values()) for k in range(5)], [6] * 5)
        for title in set(titles.values()):
            self.assertEqual(len({folds[q] for q in titles if titles[q] == title}), 1)

    def test_exact_raw_titles_are_not_normalized(self):
        titles = {1: "A", 2: "A", 3: "a", 4: " A", 5: "A ", 6: "B"}
        folds = analysis.assign_group_folds(titles, 5, 42)
        self.assertEqual(folds[1], folds[2])
        self.assertEqual(len({folds[q] for q in (1, 3, 4, 5, 6)}), 5)

    def test_group_input_is_strict(self):
        for titles in ({1: "same", 2: "same"}, {True: "A", 2: "B"}, {1: "", 2: "B"}):
            with self.assertRaises(ValueError):
                analysis.assign_group_folds(titles, 2, 42)

    def test_heldout_labels_cannot_choose_their_prompt(self):
        helper = analysis.load_helper()
        titles = {1: "one", 2: "one", 3: "two", 4: "two", 5: "three", 6: "three"}
        folds = analysis.assign_group_folds(titles, 3, 42)
        targets = {q: q % 2 for q in titles}
        probabilities = {"v0": {q: 0.4 for q in titles}, "v1": {q: 0.8 for q in titles}}
        result = analysis.strict_evaluate(helper, probabilities, targets, titles, folds, "v0")
        for fold in range(3):
            changed = {q: 1-y if folds[q] == fold else y for q, y in targets.items()}
            altered = analysis.strict_evaluate(helper, probabilities, changed, titles, folds, "v0")
            self.assertEqual(result["fold_selections"][fold]["selected_variant"],
                             altered["fold_selections"][fold]["selected_variant"])

    def test_missing_probability_and_split_title_reject(self):
        helper = analysis.load_helper()
        titles, targets = {1: "a", 2: "a", 3: "b"}, {1: 1, 2: 1, 3: 0}
        probabilities = {"v0": {1: 0.5, 2: 0.5, 3: 0.5}, "v1": {1: 0.8, 3: 0.2}}
        with self.assertRaises(ValueError):
            analysis.strict_evaluate(helper, probabilities, targets, titles, {1: 0, 2: 0, 3: 1}, "v0")
        probabilities["v1"][2] = 0.8
        with self.assertRaises(ValueError):
            analysis.strict_evaluate(helper, probabilities, targets, titles, {1: 0, 2: 1, 3: 1}, "v0")

    def test_title_weighted_scores_and_cluster_bootstrap_known_values(self):
        titles = {1: "a", 2: "a", 3: "b"}
        predictions = [{"question_id": q, "target": 1, "baseline_probability": 0.0,
                        "selected_probability": 1.0 if q < 3 else 0.0} for q in titles]
        scores = analysis.paired_summary(predictions, titles, samples=2000)
        self.assertAlmostEqual(scores["question_weighted_delta"], -2/3)
        self.assertAlmostEqual(scores["title_weighted_delta"], -0.5)
        self.assertEqual(scores["question_weighted_interval"], [-1.0, 0.0])
        self.assertEqual(scores, analysis.paired_summary(predictions, titles, samples=2000))

    def test_source_mismatch_rejects_before_execution(self):
        with self.assertRaises(ValueError):
            analysis.load_helper(expected_sha256="0" * 64)


if __name__ == "__main__":
    unittest.main()
